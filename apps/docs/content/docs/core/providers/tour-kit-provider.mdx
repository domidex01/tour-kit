Global provider that manages multiple tours across your application.

## Usage

```tsx
import { TourKitProvider } from '@tour-kit/core';

function App() {
  return (
    <TourKitProvider>
      <YourApp />
    </TourKitProvider>
  );
}
```

## With Global Config

```tsx
<TourKitProvider
  defaultConfig={{
    keyboard: true,
    spotlight: {
      padding: 8,
    },
  }}
  persistence={{
    strategy: 'localStorage',
    key: 'tour-kit-state',
  }}
>
  <App />
</TourKitProvider>
```

## Multiple Tours

When using `TourKitProvider`, you can define multiple tours:

```tsx
<TourKitProvider>
  <TourProvider tour={onboardingTour}>
    <TourProvider tour={featureTour}>
      <App />
    </TourProvider>
  </TourProvider>
</TourKitProvider>
```
